# Initialize this project from the existing GitHub repository.
mkdir -p grapplinggarage
cd grapplinggarage
git init
git remote add origin https://github.com/elegantselim/GrapplingGarage.git
git fetch
git checkout -B main origin/main
git reset --hard origin/main
npm install
npm run lint
npm run build  # static deployment bundle in out/
